georef.archivos

Rutas de trabajo, estructura de directorios y lectura del archivo de entrada.

Todas las rutas son relativas al directorio de trabajo (la raíz del proyecto), igual que .env y config/georef.toml. Los nombres de coordenadas y mapas se derivan del identificador del área: area_metropolitana o el nombre de un subconjunto.

  1"""Rutas de trabajo, estructura de directorios y lectura del archivo de entrada.
  2
  3Todas las rutas son relativas al directorio de trabajo (la raíz del proyecto), igual
  4que `.env` y `config/georef.toml`. Los nombres de coordenadas y mapas se derivan del
  5identificador del área: `area_metropolitana` o el nombre de un subconjunto.
  6"""
  7import shutil
  8import sys
  9import tkinter as tk
 10from pathlib import Path
 11from tkinter import filedialog
 12
 13import pandas as pd
 14
 15from georef.config import AREA_METROPOLITANA, Columnas
 16
 17CARPETA_ORIGINAL = Path("original")
 18CARPETA_ESTANDARIZADO = Path("estandarizado")
 19CARPETA_COORDENADAS = Path("coordenadas")
 20CARPETA_MAPAS = Path("mapas")
 21
 22# Archivos generados por la estandarización (nombres solo en ASCII)
 23RUTA_TOTAL_AM = CARPETA_ESTANDARIZADO / f"total_{AREA_METROPOLITANA}.xlsx"
 24RUTA_PRINCIPALES_AM = CARPETA_ESTANDARIZADO / f"principales_{AREA_METROPOLITANA}.xlsx"
 25RUTA_ESPECIALES_AM = CARPETA_ESTANDARIZADO / f"especiales_{AREA_METROPOLITANA}.xlsx"
 26RUTA_FUERA_AM = CARPETA_ESTANDARIZADO / f"fuera_{AREA_METROPOLITANA}.xlsx"
 27
 28# Nombre que tenía el archivo de coordenadas antes de modularizacion-config.
 29RUTA_COORDENADAS_ANTIGUA = CARPETA_COORDENADAS / "coordenadas.xlsx"
 30
 31# Prefijos de los archivos que no cuentan como archivo de entrada: ocultos (incluido el
 32# bloqueo de LibreOffice, `.~lock.<nombre>#`) y bloqueo de Excel (`~$<nombre>`).
 33PREFIJOS_IGNORADOS = (".", "~$")
 34
 35
 36def ruta_coordenadas(area: str) -> Path:
 37    """Archivo de coordenadas de un área: `coordenadas/coordenadas_<area>.xlsx`."""
 38    return CARPETA_COORDENADAS / f"coordenadas_{area}.xlsx"
 39
 40
 41def carpeta_mapas(area: str) -> Path:
 42    """Carpeta de los mapas de un área: `mapas/<area>/`."""
 43    return CARPETA_MAPAS / area
 44
 45
 46def estructura_programa():
 47    """
 48    Crea la estructura de directorios requerida por el programa si no existen.
 49    """
 50    rutas = [
 51        CARPETA_ORIGINAL,
 52        CARPETA_ESTANDARIZADO,
 53        CARPETA_COORDENADAS,
 54        carpeta_mapas(AREA_METROPOLITANA) / "filtrados",
 55    ]
 56    for ruta in rutas:
 57        ruta.mkdir(parents=True, exist_ok=True)
 58
 59
 60def archivos_de_entrada() -> list[Path]:
 61    """Archivos de `original/` que cuentan como entrada: sin subdirectorios, ocultos ni bloqueos."""
 62    if not CARPETA_ORIGINAL.is_dir():
 63        return []
 64    return sorted(
 65        ruta for ruta in CARPETA_ORIGINAL.iterdir()
 66        if ruta.is_file() and not ruta.name.startswith(PREFIJOS_IGNORADOS)
 67    )
 68
 69
 70def verificar_y_copiar_archivo_original():
 71    """
 72    Si `original/` no contiene ningún archivo de entrada, permite al usuario seleccionar uno
 73    mediante un diálogo gráfico para copiarlo a esa carpeta. Solo lo usa el menú interactivo.
 74    """
 75    if not archivos_de_entrada():
 76        print("La carpeta 'original' está vacía. Selecciona un archivo para copiar allí.")
 77
 78        # Oculta la ventana principal de tkinter
 79        root = tk.Tk()
 80        root.withdraw()
 81
 82        # Abre el diálogo para seleccionar archivo
 83        archivo_origen = filedialog.askopenfilename(title="Selecciona un archivo para copiar a 'original'")
 84
 85        if archivo_origen:
 86            # Copia el archivo a la carpeta 'original'
 87            archivo_destino = CARPETA_ORIGINAL / Path(archivo_origen).name
 88            shutil.copy(archivo_origen, archivo_destino)
 89            print(f"Archivo copiado a: {archivo_destino}")
 90        else:
 91            print("No se seleccionó ningún archivo.")
 92

 94def leer_archivo_original() -> pd.DataFrame:
 95    """
 96    Lee el único archivo de entrada de la carpeta 'original' y lo carga en un DataFrame.
 97
 98    Termina el programa con un mensaje si no hay ningún archivo de entrada o hay más de uno.
 99
100    Returns:
101        pd.DataFrame: Datos cargados desde el archivo Excel.
102    """
103    archivos = archivos_de_entrada()
104    if len(archivos) > 1:
105        sys.exit("Error: Hay más de un archivo en la carpeta 'original'. Asegúrate de que solo haya uno.")
106    if not archivos:
107        sys.exit("Error: No hay ningún archivo en la carpeta 'original'. "
108                 "Coloca allí el archivo de Excel con las direcciones.")
109    return pd.read_excel(archivos[0])

112def verificar_columnas(df: pd.DataFrame, columnas: Columnas):
113    """Termina el programa si el archivo de entrada no tiene alguna de las columnas configuradas."""
114    faltantes = [nombre for nombre in (columnas.ciudad, columnas.direccion) if nombre not in df.columns]
115    if faltantes:
116        nombres = ", ".join(f"'{nombre}'" for nombre in faltantes)
117        sys.exit(f"Error: El archivo de entrada no tiene la columna {nombres}. "
118                 "Revisa la sección [columnas] de la configuración.")

121def verificar_principales_am():
122    """
123    Termina el programa con un mensaje si no existe el archivo de direcciones principales estandarizadas,
124    necesario para la conversión de coordenadas cuando aún no hay archivo de coordenadas.
125    """
126    if not RUTA_PRINCIPALES_AM.exists():
127        sys.exit(f"Error: No se encontró '{RUTA_PRINCIPALES_AM.as_posix()}'. "
128                 "Ejecuta primero la estandarización de direcciones (opción 1).")

131def verificar_nombre_antiguo_coordenadas():
132    """
133    Termina el programa si solo existe el archivo de coordenadas con el nombre anterior.
134
135    Sin esta comprobación, la conversión no encontraría el archivo nuevo y volvería a
136    geocodificar todo desde el principio, gastando cuota de la API.
137    """
138    nueva = ruta_coordenadas(AREA_METROPOLITANA)
139    if RUTA_COORDENADAS_ANTIGUA.exists() and not nueva.exists():
140        sys.exit(f"Error: Se encontró '{RUTA_COORDENADAS_ANTIGUA.as_posix()}', el nombre anterior del archivo "
141                 f"de coordenadas. Renómbralo a '{nueva.as_posix()}' y vuelve a ejecutar "
142                 "(no hace falta volver a geocodificar).")
