 21def ciudad_consulta(ciudad: str) -> str:
 22    """Municipio y departamento de la columna de ciudad para una consulta: 'MEDELLIN - ANTIOQUIA' -> 'MEDELLIN, ANTIOQUIA'."""
 23    return ", ".join(parte.strip() for parte in str(ciudad).split(" - "))

 26def consulta_direccion(direccion: str, ciudad: str, formato: str, nombres_via) -> str:
 27    """
 28    Texto de la consulta de un registro: su vía y numeración, el municipio, el departamento y 'Colombia'.
 29
 30    Nunca incluye el complemento, el barrio ni la zona. El formato `abreviado` usa la dirección
 31    tal como la deja la estandarización (`CL 10 #20-30`); el `expandido` escribe el nombre del
 32    tipo de vía de `nombres_via` y separa `#` del número (`Calle 10 # 20-30`). La orientación
 33    pegada al número (`45S`) no se expande, para no confundirla con la letra de la vía.
 34    """
 35    if formato == "expandido":
 36        tipo, _, resto = direccion.partition(" ")
 37        direccion = " ".join(p for p in (nombres_via.get(tipo, tipo), resto.replace("#", "# ")) if p)
 38    return f"{direccion}, {ciudad_consulta(ciudad)}, Colombia"

 41def consulta_lugar(complemento, lugar_declarado, ciudad: str, palabras_lugar) -> str | None:
 42    """
 43    Texto de la segunda consulta, con el lugar con nombre del complemento, o None si no lo hay.
 44
 45    El lugar va desde la primera palabra del complemento que está en `palabras_lugar` hasta el
 46    final (`APTO 301 UNIDAD LOS PINOS` -> `UNIDAD LOS PINOS`). Le siguen el barrio o la comuna
 47    declarados, en mayúsculas, si los hay, y la ciudad.
 48    """
 49    if not isinstance(complemento, str):
 50        return None
 51    palabras = complemento.split()
 52    inicio = next((i for i, palabra in enumerate(palabras) if palabra.upper() in palabras_lugar), None)
 53    if inicio is None:
 54        return None
 55    partes = [" ".join(palabras[inicio:])]
 56    if isinstance(lugar_declarado, str) and lugar_declarado.strip():
 57        partes.append(lugar_declarado.upper())
 58    return ", ".join(partes + [ciudad_consulta(ciudad), "Colombia"])

 61def consulta_municipio(municipio: str) -> str:
 62    """Texto de la consulta del punto de un municipio: '<MUNICIPIO>, <DEPARTAMENTO>, Colombia'."""
 63    return f"{ciudad_consulta(municipio)}, Colombia"

 99class SesionConsultas:
100    """
101    Consultas a OpenCage de una ejecución, con pausa, caché, registro de diagnóstico y paradas.
102
103    - Antes de cada petición espera hasta que hayan pasado `pausa_segundos` desde el inicio de
104      la anterior, también si la anterior terminó en error.
105    - Cada respuesta correcta (con o sin resultados) se añade a la caché en cuanto llega, y una
106      consulta que ya está en la caché nunca se repite.
107    - Deja de hacer peticiones, sin perder lo obtenido, si OpenCage indica cuota o ritmo excedidos,
108      tras `MAXIMO_ERRORES_SEGUIDOS` errores seguidos o al llegar a `limite_consultas`. Desde ese
109      momento, lo que no está en la caché queda PENDIENTE.
110    - Sin llave (`llave=None`) no hace peticiones: solo usa la caché.
111    - Sin `ruta_cache` (None), las respuestas solo se recuerdan durante la sesión.
112    - Si OpenCage rechaza la llave, termina el programa sin mostrarla.
113    """
114
115    def __init__(self, llave, parametros, ruta_cache, ruta_registro, operacion):
116        self._cliente = OpenCageGeocode(llave) if llave else None
117        self._pausa = parametros.pausa_segundos
118        self._limite = parametros.limite_consultas
119        self._ruta_cache = Path(ruta_cache) if ruta_cache is not None else None
120        self._ruta_registro = Path(ruta_registro)
121        self._operacion = operacion
122        self._cache = _leer_cache(self._ruta_cache) if self._ruta_cache is not None else {}
123        self._inicio_anterior = None
124        self._errores_seguidos = 0
125        self.peticiones = 0
126        self.desde_cache = 0
127        self.pendientes = set()  # consultas distintas que quedaron sin respuesta
128        self.parada = None
129        self.restantes = None  # últimas peticiones restantes que informó OpenCage
130
131    def resolver(self, consulta, bounds=None, registro="") -> Respuesta:
132        """Devuelve la respuesta a `consulta`, de la caché o con una petición. `registro` va al registro."""
133        if consulta in self._cache:
134            self.desde_cache += 1
135            return self._cache[consulta]
136        if self._cliente is None or self.parada:
137            self.pendientes.add(consulta)
138            return RESPUESTA_PENDIENTE
139        if self.peticiones >= self._limite:
140            self.parada = PARADA_LIMITE
141            self.pendientes.add(consulta)
142            return RESPUESTA_PENDIENTE
143
144        parametros = {"countrycode": "co", "limit": 1, "no_annotations": 1}
145        if bounds:
146            parametros["bounds"] = bounds
147        self._esperar()
148        self._inicio_anterior = time.monotonic()
149        self.peticiones += 1
150        try:
151            datos = self._cliente.geocode(consulta, raw_response=True, **parametros)
152        except RateLimitExceededError:
153            self._registrar(registro, "RateLimitExceededError")
154            self.parada = PARADA_CUOTA
155            self.pendientes.add(consulta)
156            return RESPUESTA_PENDIENTE
157        except (NotAuthorizedError, ForbiddenError) as error:
158            self._registrar(registro, type(error).__name__)
159            sys.exit("Error: OpenCage rechazó la llave de 'OPENCAGE_API_KEY' (no autorizada o bloqueada). "
160                     "Revisa la llave en tu archivo '.env' o en el entorno. No se escribió el archivo de coordenadas.")
161        except (UnknownError, InvalidInputError, requests.RequestException) as error:
162            self._registrar(registro, type(error).__name__)
163            self._errores_seguidos += 1
164            if self._errores_seguidos >= MAXIMO_ERRORES_SEGUIDOS:
165                self.parada = PARADA_ERRORES
166            self.pendientes.add(consulta)
167            return RESPUESTA_PENDIENTE
168
169        self._errores_seguidos = 0
170        self.restantes = (datos.get("rate") or {}).get("remaining", self.restantes)
171        respuesta = _primera_respuesta(datos.get("results") or [])
172        self._registrar(registro, respuesta.estado, respuesta, (datos.get("rate") or {}).get("remaining"))
173        self._guardar_en_cache(consulta, respuesta)
174        return respuesta
175
176    def _esperar(self):
177        if self._inicio_anterior is not None:
178            espera = self._pausa - (time.monotonic() - self._inicio_anterior)
179            if espera > 0:
180                time.sleep(espera)
181
182    def _guardar_en_cache(self, consulta, respuesta):
183        self._cache[consulta] = respuesta
184        if self._ruta_cache is None:
185            return
186        fila = {"consulta": consulta, "fecha": datetime.date.today().isoformat(), "estado": respuesta.estado,
187                "latitud": respuesta.latitud, "longitud": respuesta.longitud, "confianza": respuesta.confianza,
188                "tipo": respuesta.tipo, "texto": respuesta.texto}
189        _anadir_fila(self._ruta_cache, COLUMNAS_CACHE, fila)
190
191    def _registrar(self, registro, resultado, respuesta=None, restantes=None):
192        duracion = round((time.monotonic() - self._inicio_anterior) * 1000)
193        fila = {"fecha_hora_utc": datetime.datetime.now(datetime.timezone.utc).isoformat(timespec="seconds"),
194                "operacion": self._operacion, "registro": registro, "duracion_ms": duracion,
195                "resultado": resultado, "restantes": restantes,
196                "tipo": respuesta.tipo if respuesta else None,
197                "confianza": respuesta.confianza if respuesta else None}
198        _anadir_fila(self._ruta_registro, COLUMNAS_REGISTRO, fila)
199
200
201def _primera_respuesta(resultados) -> Respuesta:
202    if not resultados:
203        return Respuesta(SIN_RESULTADO)
204    primero = resultados[0]
205    return Respuesta(
206        RESULTADO,
207        latitud=float(primero["geometry"]["lat"]),
208        longitud=float(primero["geometry"]["lng"]),
209        confianza=primero.get("confidence"),
210        tipo=(primero.get("components") or {}).get("_type"),
211        texto=primero.get("formatted"),
212    )

215def _leer_cache(ruta: Path) -> dict:
216    """Carga la caché CSV en un diccionario consulta -> Respuesta (vacío si el archivo no existe)."""
217    if not ruta.exists():
218        return {}
219    cache = {}
220    with open(ruta, encoding="utf-8", newline="") as archivo:
221        for fila in csv.DictReader(archivo):
222            def valor(columna, tipo):
223                texto = fila.get(columna) or ""
224                return tipo(texto) if texto else None
225            cache[fila["consulta"]] = Respuesta(
226                fila["estado"], valor("latitud", float), valor("longitud", float),
227                valor("confianza", lambda t: int(float(t))), fila.get("tipo") or None, fila.get("texto") or None,
228            )
229    return cache

232def _anadir_fila(ruta: Path, columnas, fila):
233    """Añade `fila` al CSV de `ruta` (con encabezado si es nuevo) y lo cierra, para no perderla si el proceso cae."""
234    nuevo = not ruta.exists() or ruta.stat().st_size == 0
235    ruta.parent.mkdir(parents=True, exist_ok=True)
236    with open(ruta, "a", encoding="utf-8", newline="") as archivo:
237        escritor = csv.DictWriter(archivo, fieldnames=columnas)
238        if nuevo:
239            escritor.writeheader()
240        escritor.writerow({c: "" if fila.get(c) is None else fila[c] for c in columnas})

243def obtener_llave():
244    """
245    Obtiene la llave API de OpenCage de la variable 'OPENCAGE_API_KEY'.
246
247    La busca primero en el entorno del proceso y, si ahí no está o está vacía, en el archivo '.env' del
248    directorio de trabajo, sin modificar el entorno. La llave nunca se guarda en archivos ni se solicita
249    por consola. Si no se encuentra, el programa termina con un mensaje que explica cómo definirla.
250
251    Returns:
252        str: La llave API sin espacios al inicio ni al final.
253    """
254    llave = os.environ.get("OPENCAGE_API_KEY", "").strip()
255    if not llave:
256        llave = (dotenv_values(".env").get("OPENCAGE_API_KEY") or "").strip()
257    if not llave:
258        sys.exit(
259            "Error: No se encontró la llave de OpenCage en la variable 'OPENCAGE_API_KEY'.\n"
260            "Defínela en un archivo '.env' en la carpeta del proyecto:\n"
261            "  cp .env.example .env    (luego escribe la llave en .env y ejecuta: chmod 600 .env)\n"
262            "o como variable de entorno, por ejemplo:\n"
263            "  export OPENCAGE_API_KEY=<tu-llave>"
264        )
265    return llave

141            self.parada = PARADA_LIMITE
142            self.pendientes.add(consulta)
143            return RESPUESTA_PENDIENTE
144
145        parametros = {"countrycode": "co", "limit": 1, "no_annotations": 1}
146        if bounds:
147            parametros["bounds"] = bounds
148        self._esperar()
149        self._inicio_anterior = time.monotonic()
150        self.peticiones += 1
151        try:
152            datos = self._cliente.geocode(consulta, raw_response=True, **parametros)
153        except RateLimitExceededError:

156            self.pendientes.add(consulta)

161                     "Revisa la llave en tu archivo '.env' o en el entorno. No se escribió el archivo de coordenadas.")
162        except (UnknownError, InvalidInputError, requests.RequestException) as error:

164            self._errores_seguidos += 1
165            if self._errores_seguidos >= MAXIMO_ERRORES_SEGUIDOS:
166                self.parada = PARADA_ERRORES
